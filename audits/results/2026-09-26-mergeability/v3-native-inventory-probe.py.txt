"""Measure retained native market inventories at adjacent historical blocks."""
import asyncio
import hashlib
import json
import os
import resource
import sys
import tomllib
import tracemalloc
from dataclasses import asdict
from collections import Counter
from pathlib import Path
from time import monotonic
from typing import Any

from common import write_json


def cache_metrics() -> dict[str, Any]:
    from y.prices._routing import quote_service
    from y.prices._rpc import state_cache

    service = quote_service()
    inventories = service.market_cache.values
    return {
        "market_keys": len(inventories),
        "market_cache_capacity": inventories.maxsize,
        "inventory_market_references": sum(len(value) for value in inventories.values()),
        "inventory_tuple_shallow_bytes": sum(sys.getsizeof(value) for value in inventories.values()),
        "result_keys": len(service.result_cache.values),
        "state_keys": len(state_cache().values),
        "rss_peak_bytes": resource.getrusage(resource.RUSAGE_SELF).ru_maxrss * 1024,
        "cgroup_current_bytes": int(Path('/sys/fs/cgroup/memory.current').read_text()),
        "traced_current_bytes": tracemalloc.get_traced_memory()[0],
    }


async def monitor(directory: Path, started: float) -> None:
    with (directory / 'retention-progress.jsonl').open('w') as stream:
        while True:
            row = {"elapsed_seconds": monotonic() - started, **cache_metrics()}
            if tracemalloc.is_tracing():
                snapshot = tracemalloc.take_snapshot()
                row['largest_allocations'] = [
                    {'location': str(stat.traceback), 'size': stat.size, 'count': stat.count}
                    for stat in snapshot.statistics('lineno')[:25]
                ]
                del snapshot
            stream.write(json.dumps(row) + '\n')
            stream.flush()
            await asyncio.sleep(60)


async def verify_v3_indexes(token: str, number: int) -> list[dict[str, Any]]:
    from y import convert
    from y.prices.dex.uniswap import uniswap_multiplexer
    from y.prices._rpc import BlockRef, deployed

    token = await convert.to_address_async(token)
    block = await BlockRef.resolve(number)
    results = []
    for router in [uniswap_multiplexer.v3, *uniswap_multiplexer.v3_forks]:
        if router is None or not await deployed(str(router._factory).lower(), block):
            continue
        registry = await router.__pools__
        for requested in [number, number - 1, number]:
            # Factory metadata is the independent source of membership; invoking
            # the public iterator here bypasses the market snapshot cache.
            expected = [p.address for p in registry._objects if p._deploy_block <= requested and token in p]
            actual = [p.address async for p in router.pools_for_token(token, requested)]
            assert actual == expected, (str(router._factory), requested, len(expected), len(actual))
            assert len(actual) == len(set(actual)), (str(router._factory), requested, 'duplicates')
            results.append({'factory': str(router._factory), 'block': requested, 'pools': len(actual), 'sha256': hashlib.sha256(json.dumps(actual).encode()).hexdigest(), 'exact_metadata_match': True, 'unique': True})
    await block.verify()
    return results


async def main() -> int:
    from y.prices._routing import quote_service
    from y.prices._rpc import BlockRef

    directory = Path(os.environ['VALIDATION_REPORT'])
    deadline = tomllib.loads(Path('pyproject.toml').read_text())['tool']['pytest']['ini_options']['asyncio_task_timeout']
    token = '0xc02aaa39b223fe8d0a0e5c4f27ead9083c756cc2'
    rows: list[dict[str, Any]] = []
    started = monotonic()
    sampler = asyncio.create_task(monitor(directory, started))
    complete = False
    try:
        for phase, number in [('cold', 23479243), ('adjacent', 23479244), ('repeat', 23479243)]:
            begin = monotonic()
            row: dict[str, Any] = {'phase': phase, 'token': token, 'block': number, 'deadline_seconds': deadline}
            try:
                block = await BlockRef.resolve(number)
                row['block_hash'] = block.hash
                markets = await asyncio.wait_for(quote_service().markets(token, block, False), deadline)
                await block.verify()
                digest = hashlib.sha256()
                for market in markets:
                    digest.update(json.dumps(asdict(market), sort_keys=True, default=str).encode())
                    digest.update(b'\n')
                identities = [(m.factory, m.pool) for m in markets if m.protocol in ('Uniswap V3', 'Slipstream')]
                assert len(identities) == len(set(identities)), 'duplicate V3 market snapshots'
                if phase == 'cold':
                    assert block.hash == '0x0b6c2373db136669e187029cb13c2893a69d9e5d6bcc97044d38e83123cf5a76'
                    assert digest.hexdigest() == '05e59e50da97a1151a29497ae18a7e49c02e38fb7adbe509ec162516c6409f98', 'cold inventory differs from native baseline'
                row.update(status='pass', canonical_verified=True, markets=len(markets), inventory_sha256=digest.hexdigest(), protocol_counts=dict(Counter(m.protocol for m in markets)), v3_unique=True)
                row['v3_indexes'] = await asyncio.wait_for(verify_v3_indexes(token, number), deadline)
                del markets
                row.update(cache_metrics())
            except Exception as exc:
                row.update(status='failure', error=f'{type(exc).__name__}: {exc}', **cache_metrics())
            row['elapsed_seconds'] = monotonic() - begin
            rows.append(row)
            write_json(directory / 'retention-cases.json', rows)
            print(json.dumps(row), flush=True)
            if row['status'] != 'pass':
                break
        complete = len(rows) == 3 and all(row['status'] == 'pass' for row in rows)
        if complete:
            assert rows[0]['inventory_sha256'] == rows[2]['inventory_sha256']
            assert rows[0]['block_hash'] == rows[2]['block_hash']
        return 0 if complete else 1
    finally:
        sampler.cancel()
        await asyncio.gather(sampler, return_exceptions=True)
        write_json(directory / 'retention-summary.json', {'complete': complete, 'elapsed_seconds': monotonic() - started, 'cache_metrics': cache_metrics(), 'instrumentation': 'Lightweight cache metrics only; no tracemalloc, object census, or SQLite backup. Exact cold baseline digest plus unique V3 snapshots and direct warm index checks against factory metadata. Adjacent baseline digest contained duplicate V3 pools and is not an expected result.'})


if __name__ == '__main__':
    from brownie._config import CONFIG
    from brownie.network.main import connect

    connect(os.environ['BROWNIE_NETWORK'])
    CONFIG.settings['autofetch_sources'] = False
    raise SystemExit(asyncio.get_event_loop().run_until_complete(main()))
